async function loadData() { const tBody = document.getElementById('giderTabloBody'); try { const response = await fetch("/api/login", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "getData" }) }); const res = await response.json(); if(res && res.ok && res.data) { tBody.innerHTML = ''; if (res.data.length === 0) { tBody.innerHTML = 'Henüz kaydedilmiş harcama bulunamadı.'; return; } let toplam = 0; res.data.forEach(g => { toplam += Number(g.purchase_total || 0); tBody.innerHTML += ` ${g.period || '-'} ${g.date || '-'} ${g.description || '-'} ${g.category || '-'} ${g.cash ? 'Nakit' : 'Kart'} ₺${Number(g.purchase_total || 0).toLocaleString('tr-TR')} `; }); document.getElementById('kpiSeciliDonem').innerText = `₺${toplam.toLocaleString('tr-TR')}`; } else { tBody.innerHTML = 'Veritabanından yanıt alınamadı.'; } } catch(e) { tBody.innerHTML = 'Veritabanı bağlantı hatası! Lütfen API köprünüzü ve Supabase adresinizi kontrol edin.'; } }
