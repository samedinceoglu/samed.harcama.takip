const SPREADSHEET_ID = "1wwW48GDy5sssSMO1o7j1U_qBLiklfGU7og1lYd-2Duo"; // Harcama Takip Veritabanı ID

function doGet(e) {
  return HtmlService.createTemplateFromFile('Index')
    .evaluate()
    .setTitle('Harcama Takip Uygulaması')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function getVeriler() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Sayfa1');
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  
  return data.map((row, index) => ({
    rowId: index + 2,
    id: row[0],
    period: row[1],
    date: row[2],
    description: row[3],
    category: row[4],
    credit: row[6] || 0,
    cash: row[11] || 0,
    detail: row[12] || ""
  }));
}

function harcamaEkle(harcama) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('Sayfa1');
  const id = 'custom-' + new Date().getTime();
  sheet.appendRow([
    id,
    harcama.date.substring(0, 7),
    harcama.date,
    harcama.description,
    harcama.category,
    1,
    harcama.credit || 0,
    0, 0, 0, 0,
    harcama.cash || 0,
    harcama.detail || ""
  ]);
  return { status: "success", id: id };
}
